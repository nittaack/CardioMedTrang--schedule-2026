    function saveLeave() {
      const name = document.getElementById('docName').value;
      const start = document.getElementById('startDate').value;
      const end = document.getElementById('endDate').value;

      if(!start || !end) {
        alert('กรุณาเลือกวันที่เริ่มและสิ้นสุดวันลา');
        return;
      }

      let endDateObj = new Date(end);
      endDateObj.setDate(endDateObj.getDate() + 1);
      let formattedEnd = endDateObj.toISOString().split('T')[0];

      calendar.addEvent({
        title: "🔴 [ลา] " + name,
        start: start,
        end: formattedEnd,
        backgroundColor: "#E53935", // วันลาเป็นสีแดงสดเสมอ
        priority: 1
      });

      closeModal();
    }
